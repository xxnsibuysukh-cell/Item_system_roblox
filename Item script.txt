local FAST = TweenInfo.new(0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
local SMOOTH = TweenInfo.new(0.3, Enum.EasingStyle.Quint, Enum.EasingDirection.Out)
local SPRING = TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
local PRESS = TweenInfo.new(0.08, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
local FADE = TweenInfo.new(0.2, Enum.EasingStyle.Sine, Enum.EasingDirection.Out)

local function tween(obj, info, props)
	local t = TweenService:Create(obj, info, props)
	t:Play()
	return t
end

local function new(className, props, parent)
	local inst = Instance.new(className)
	for key, value in pairs(props) do
		inst[key] = value
	end
	inst.Parent = parent
	return inst
end

local function round(parent, radius)
	return new("UICorner", { CornerRadius = UDim.new(0, radius) }, parent)
end

local function stroke(parent, color, thickness, transparency)
	return new("UIStroke", {
		Color = color,
		Thickness = thickness or 1,
		Transparency = transparency or 0,
		ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
	}, parent)
end

local function gradient(parent, c1, c2, rotation)
	return new("UIGradient", { Color = ColorSequence.new(c1, c2), Rotation = rotation or 0 }, parent)
end

local function addButtonFx(button, base, hover)
	button.AutoButtonColor = false
	button.BackgroundColor3 = base
	local scale = new("UIScale", { Scale = 1 }, button)
	button.MouseEnter:Connect(function()
		tween(button, FAST, { BackgroundColor3 = hover })
	end)
	button.MouseLeave:Connect(function()
		tween(button, FAST, { BackgroundColor3 = base })
		tween(scale, FAST, { Scale = 1 })
	end)
	button.MouseButton1Down:Connect(function()
		tween(scale, PRESS, { Scale = 0.94 })
	end)
	button.MouseButton1Up:Connect(function()
		tween(scale, SPRING, { Scale = 1 })
	end)
end

local function makeButton(parent, text, size, position, base, hover, font, textSize, order)
	local b = new("TextButton", {
		Size = size,

		LayoutOrder = order or 0,
	}, parent)
	round(b, 7)
	addButtonFx(b, base or THEME.button, hover or THEME.buttonHover)

end

local function isPress(input)
	return input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch
end

local function isMove(input)
	return input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch
end

local function makeDraggable(handle, target, onTap)
	local dragging, moved = false, false
	local startMouse, startPos
	handle.InputBegan:Connect(function(input)
		if isPress(input) then
			dragging, moved = true, false
			startMouse = input.Position
			startPos = target.Position
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
					if not moved and onTap then
						onTap()
					end
				end
			end)
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if dragging and isMove(input) then
			local d = input.Position - startMouse
			if d.Magnitude > 4 then
				moved = true
			end
			if moved then
				local cam = Workspace.CurrentCamera
				local vp = cam and cam.ViewportSize or Vector2.new(1920, 1080)
				local sz = target.AbsoluteSize
				local x = math.clamp(startPos.X.Offset + d.X, 0, math.max(vp.X - math.min(sz.X, 60), 0))
				local y = math.clamp(startPos.Y.Offset + d.Y, 0, math.max(vp.Y - math.min(sz.Y, 40), 0))
				target.Position = UDim2.fromOffset(x, y)
			end
		end
	end)
end

		Position = UDim2.new(0, default and 20 or 2, 0.5, 0),
		BackgroundColor3 = THEME.text,
		BorderSizePixel = 0,
	}, sw)
	round(knob, 6)
	local state = default
	local api = {}
	function api.set(v, silent)
		state = v
		tween(sw, FAST, { BackgroundColor3 = v and THEME.accent or THEME.off })
		tween(knob, FAST, { Position = UDim2.new(0, v and 20 or 2, 0.5, 0) })
		if not silent then
			onChange(v)
		end
	end
	function api.get()
		return state
	end
	sw.MouseButton1Click:Connect(function()
		api.set(not state)
	end)
	return api, row
end

local function makeSlider(parent, order, text, min, max, value, step, fmt, onChange)
	local frame = new("Frame", {
		Size = UDim2.new(1, 0, 0, 38),
		BackgroundColor3 = THEME.card,
		BorderSizePixel = 0,
		LayoutOrder = order,
	}, parent)
	round(frame, 7)
	new("TextLabel", {
		Size = UDim2.new(0.6, 0, 0, 18),
		Position = UDim2.fromOffset(10, 3),
		BackgroundTransparency = 1,
		Font = Enum.Font.GothamMedium,
		Text = text,
		TextColor3 = THEME.subtext,
		TextSize = 10,
		TextXAlignment = Enum.TextXAlignment.Left,
	}, frame)
	local valueLabel = new("TextLabel", {
		Size = UDim2.new(0.4, -10, 0, 18),
		Position = UDim2.new(0.6, 0, 0, 3),
		BackgroundTransparency = 1,
		Font = Enum.Font.GothamBold,
		Text = "",
		TextColor3 = THEME.text,
		TextSize = 10,
		TextXAlignment = Enum.TextXAlignment.Right,
	}, frame)
	local track = new("TextButton", {
		Size = UDim2.new(1, -20, 0, 6),
		Position = UDim2.new(0, 10, 0, 26),
		BackgroundColor3 = THEME.header,
		Text = "",
		AutoButtonColor = false,
		BorderSizePixel = 0,
	}, frame)
	round(track, 3)
	local fill = new("Frame", {
		Size = UDim2.fromScale(0, 1),
		BackgroundColor3 = THEME.text,
		BorderSizePixel = 0,
	}, track)
	round(fill, 3)
	gradient(fill, THEME.accent, THEME.accent2, 0)
	local knob = new("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5),
		Size = UDim2.fromOffset(14, 14),
		Position = UDim2.fromScale(0, 0.5),
		BackgroundColor3 = THEME.text,
		BorderSizePixel = 0,
		ZIndex = 2,
	}, track)
	round(knob, 7)
	stroke(knob, THEME.accent, 2)

	local current = value
	local dragging = false
	local function render(v)
		local a = (v - min) / (max - min)
		valueLabel.Text = string.format(fmt, v)
		fill.Size = UDim2.fromScale(a, 1)
		knob.Position = UDim2.fromScale(a, 0.5)
	end
	render(value)

	local function setFromX(x)
		local a = math.clamp((x - track.AbsolutePosition.X) / math.max(track.AbsoluteSize.X, 1), 0, 1)
		local v = min + a * (max - min)
		v = math.clamp(math.floor(v / step + 0.5) * step, min, max)
		if v ~= current then
			current = v
			render(v)
			onChange(v)
		end
	end
	track.InputBegan:Connect(function(input)
		if isPress(input) then
			dragging = true
			setFromX(input.Position.X)
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if dragging and isMove(input) then
			setFromX(input.Position.X)
		end
	end)
	UserInputService.InputEnded:Connect(function(input)
		if isPress(input) then
			dragging = false
		end
	end)

	local api = { frame = frame }
	function api.set(v)
		current = math.clamp(v, min, max)
		render(current)
	end
	return api
end

local desiredStatus = "Idle\nSelecting is off"
local statusToken = 0

local function setStatus(text)
	if text == desiredStatus then
		return
	end
	desiredStatus = text
	statusToken += 1
	local token = statusToken
	task.spawn(function()
		local out = tween(statusLabel, FAST, { TextTransparency = 1 })
		out.Completed:Wait()
		if token ~= statusToken then
			return
		end
		statusLabel.Text = text
		tween(statusLabel, FAST, { TextTransparency = 0 })
	end)
end

refreshStatus = function()
	local n, m = selectedCount, #luckyParts
	local text
	if not scriptOn then
		text = "Script is OFF\nTurn on power to continue"
	elseif currentIndex > 0 and luckyParts[currentIndex] then
		text = ("Similar %d of %d\n%s"):format(currentIndex, m, shortName(luckyParts[currentIndex].Name))
	elseif luckySearched then
		if m > 0 then
			text = ("%d selected · %d similar\nUse arrows to teleport"):format(n, m)
		else
			text = ("%d selected\nNo similar objects found"):format(n)
		end
	elseif mode == "Idle" then
		text = n == 0 and "Idle\nSelecting is off" or ("%d selected\nIdle · selecting is off"):format(n)
	else
		text = n == 0 and "Select mode\nClick objects to pick" or ("%d selected\nClick again to deselect"):format(n)
	end
	setStatus(text)
	local barColor = (not scriptOn) and THEME.danger or MODE_STYLE[mode].color
	tween(statusBar, FAST, { BackgroundColor3 = barColor })
end

notify = function(text, neutral)
	if not neutral then
		statusCard.BackgroundColor3 = THEME.error
		tween(statusCard, TweenInfo.new(0.6, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { BackgroundColor3 = THEME.header })
	end
	setStatus(text)
	local token = statusToken
	task.delay(2.2, function()
		if statusToken == token then
			refreshStatus()
		end
	end)
end

local function setMode(newMode)
	if newMode == "Select" and not scriptOn then
		notify("Turn on Script power first")
		return
	end
	mode = newMode
	local style = MODE_STYLE[newMode]
	tween(indicator, SMOOTH, { Position = style.pos, BackgroundColor3 = style.color })
	tween(idleButton, FAST, { TextColor3 = newMode == "Idle" and style.text or THEME.inactive })
	tween(selectButton, FAST, { TextColor3 = newMode == "Select" and style.text or THEME.inactive })
	refreshStatus()
end

local function createEsp(part, color)
	local hl = Instance.new("Highlight")
	hl.Name = "IH_ESP"
	hl.Adornee = part
	hl.FillColor = color
	hl.OutlineColor = color
	hl.FillTransparency = S.espFill
	hl.OutlineTransparency = 0
	hl.DepthMode = S.espWalls and Enum.HighlightDepthMode.AlwaysOnTop or Enum.HighlightDepthMode.Occluded
	hl.Parent = part

	local bb = Instance.new("BillboardGui")
	bb.Name = "IH_LABEL"
	bb.Adornee = part
	bb.AlwaysOnTop = true
	bb.Size = UDim2.fromOffset(150, 32)
	bb.StudsOffsetWorldSpace = Vector3.new(0, part.Size.Y / 2 + 2, 0)
	bb.LightInfluence = 0
	bb.Parent = part
	local txt = new("TextLabel", {
		Size = UDim2.fromScale(1, 1),

		TextColor3 = color,
		TextStrokeTransparency = 0.3,
		TextStrokeColor3 = Color3.new(0, 0, 0),
		Text = shortName(part.Name),
	}, bb)
	return hl, bb, txt
end

local onLost

local function selectPart(part, color)
	if selected[part] or not part or not part.Parent then
		return false
	end
	color = color or pickColor
	selOrder += 1
	local hl, bb, txt = createEsp(part, color)
	local data = { part = part, hl = hl, bb = bb, txt = txt, color = color, order = selOrder }
	data.conn = part.AncestryChanged:Connect(function()
		if not part:IsDescendantOf(Workspace) then
			onLost(part)
		end
	end)
	selected[part] = data
	selectedCount += 1
	syncLuckyVisibility(part)
	scheduleRefresh()
	if selectedCount == 32 then
		notify("Highlight limit reached (~31)\nLabels still show for all", true)
	end
	return true
end

local function deselectPart(part)
	local d = selected[part]
	if not d then
		return nil
	end
	selected[part] = nil
	selectedCount -= 1
	if d.conn then d.conn:Disconnect() end
	if d.hl then d.hl:Destroy() end
	if d.bb then d.bb:Destroy() end
	if marked[part] then
		marked[part] = nil
		markedCount -= 1
	end
	tpChecked[part] = nil
	syncLuckyVisibility(part)
	scheduleRefresh()
	return d.color
end

onLost = function(part)
	local d = selected[part]
	if not d then
		return
	end
	if S.autoRefind then
		table.insert(pending, {
			Name = part.Name,
			ClassName = part.ClassName,
			Size = part.Size,
			Position = part.Position,
			Color = d.color,
		})
		if #pending > 300 then
			table.remove(pending, 1)
		end
	end
	deselectPart(part)
end

local function matchesSig(inst, sig)
	return inst.Name == sig.Name
		and inst.ClassName == sig.ClassName
		and (inst.Size - sig.Size).Magnitude < 0.25
		and (inst.Position - sig.Position).Magnitude <= 8
end

local function tryRefind(inst)
	if selected[inst] then
		return false
	end
	for i = #pending, 1, -1 do
		if matchesSig(inst, pending[i]) then
			local color = pending[i].Color
			table.remove(pending, i)
			return selectPart(inst, color)
		end
	end
	return false
end

Workspace.DescendantAdded:Connect(function(inst)
	if not S.autoRefind or #pending == 0 or not inst:IsA("BasePart") then
		return
	end
	task.delay(0.12, function()
		if inst.Parent and #pending > 0 then
			tryRefind(inst)
		end
	end)
end)

local function setColor(part, color)
	local d = selected[part]
	if not d then
		return
	end
	d.color = color
	d.hl.FillColor = color
	d.hl.OutlineColor = color
	d.txt.TextColor3 = color
end

		return
	end
	table.insert(undoStack, entry)
	if #undoStack > HISTORY_LIMIT then
		table.remove(undoStack, 1)
	end
	redoStack = {}
	updateHistoryButtons()
end

local function applyEntry(entry, reverse)
	local toAdd = reverse and entry.removed or entry.added
	local toRemove = reverse and entry.added or entry.removed
	for _, it in ipairs(toRemove) do deselectPart(it.part) end
	for _, it in ipairs(toAdd) do selectPart(it.part, it.color) end
end

undoButton.MouseButton1Click:Connect(function()
	local entry = table.remove(undoStack)
	if not entry then return end
	applyEntry(entry, true)
	table.insert(redoStack, entry)
	updateHistoryButtons()
end)
redoButton.MouseButton1Click:Connect(function()
	local entry = table.remove(redoStack)
	if not entry then return end
	applyEntry(entry, false)
	table.insert(undoStack, entry)
	updateHistoryButtons()
end)

		if c then

		end
	end
	pushHistory(entry)
end

local function fadeInBox(box, animate)
	if animate then
		box.SurfaceTransparency = 1
		box.Transparency = 1

	else
		box.SurfaceTransparency = 0.8

	end
end

local function clearLucky()
	local animate = #luckyMarkers <= ANIMATE_LIMIT
	for _, box in ipairs(luckyMarkers) do
		if animate and box.Parent then
			local t = tween(box, FADE, { SurfaceTransparency = 1, Transparency = 1 })
			t.Completed:Once(function() box:Destroy() end)
		else
			box:Destroy()
		end
	end
	luckyMarkers, luckyParts, luckyMap = {}, {}, {}
	currentIndex = 0
	luckySearched = false
	updateNav()
	scheduleRefresh()
end

local function sortedSize(p)
	local s = p.Size
	local t = { s.X, s.Y, s.Z }
	table.sort(t)
	return t
end

local function runLucky()
	if luckyBusy then return end
	if not scriptOn then notify("Turn on Script power first") return end
	if selectedCount == 0 then notify("Select something first") return end
	luckyBusy = true
	clearLucky()

	local refs = {}
	for part in pairs(selected) do
		table.insert(refs, {
			cls = part.ClassName,
			name = part.Name,
			mesh = part:IsA("MeshPart") and part.MeshId or nil,
			mat = part.Material,
			col = part.Color,
			sz = sortedSize(part),
		})
	end
	local chars = {}
	for _, pl in ipairs(Players:GetPlayers()) do
		if pl.Character then table.insert(chars, pl.Character) end
	end
	local tol = S.similarTol

	local found = {}
	local all = Workspace:GetDescendants()
	for i, inst in ipairs(all) do
		if inst:IsA("BasePart") and not inst:IsA("Terrain") and not selected[inst] then
			for _, ref in ipairs(refs) do
				if inst.ClassName == ref.cls
					and (inst.Name == ref.name or (inst.Material == ref.mat and inst.Color == ref.col))
					and (not ref.mesh or inst.MeshId == ref.mesh)
				then
					local sz = sortedSize(inst)
					if math.abs(sz[1] - ref.sz[1]) <= tol and math.abs(sz[2] - ref.sz[2]) <= tol and math.abs(sz[3] - ref.sz[3]) <= tol then
						local inChar = false
						for _, c in ipairs(chars) do
							if inst:IsDescendantOf(c) then inChar = true break end
						end
						if not inChar then
							table.insert(found, inst)
						end
						break
					end
				end
			end
			if #found >= MAX_LUCKY then break end
		end
		if i % 4000 == 0 then task.wait() end
	end

	local hrp = getHRP()
	if hrp then
		local origin = hrp.Position
		table.sort(found, function(a, b)
			return (a.Position - origin).Magnitude < (b.Position - origin).Magnitude
		end)
	end

	local animate = #found <= ANIMATE_LIMIT
	for _, part in ipairs(found) do
		local box = makeBox(part, LUCKY_COLOR)
		fadeInBox(box, animate)
		table.insert(luckyMarkers, box)
		luckyMap[part] = box
		syncLuckyVisibility(part)
	end
	luckyParts = found
	luckySearched = true
	currentIndex = 0
	luckyBusy = false
	updateNav()
	scheduleRefresh()
	refreshStatus()
end
luckyButton.MouseButton1Click:Connect(runLucky)

local function gotoSimilar(step)
	local n = #luckyParts
	if n == 0 then
		notify("Press I'm Feeling Lucky! first")
		return
	end
	if currentIndex == 0 and step < 0 then currentIndex = 1 end
	for _ = 1, n do
		currentIndex = ((currentIndex - 1 + step) % n) + 1
		local p = luckyParts[currentIndex]
		if p and p.Parent then
			teleportTo(p)
			break
		end
	end
	updateNav()
	refreshStatus()
end
leftButton.MouseButton1Click:Connect(function() gotoSimilar(-1) end)
rightButton.MouseButton1Click:Connect(function() gotoSimilar(1) end)

teleportTo = function(part)
	local hrp = getHRP()
	if not hrp then
		notify("Character not found")
		return nil
	end
	if not part or not part.Parent then
		notify("Object no longer exists")
		return nil
	end
	if tpTween then tpTween:Cancel() end
	local pos = part.Position + Vector3.new(0, part.Size.Y / 2 + 3 + S.tpHeight, 0)
	local goal = CFrame.new(pos) * (hrp.CFrame - hrp.CFrame.Position)
	hrp.AssemblyLinearVelocity = Vector3.zero
	if S.glideTime <= 0.01 then
		hrp.CFrame = goal
		return nil
	end
	tpTween = TweenService:Create(hrp, TweenInfo.new(S.glideTime, Enum.EasingStyle.Quint, Enum.EasingDirection.InOut), { CFrame = goal })
	tpTween:Play()
	return tpTween
end

local function stopTour()
	tourRunning = false
	tourButton.Text = "▶  Start tour (checked)"
end

tourButton.MouseButton1Click:Connect(function()
	if tourRunning then
		stopTour()
		return
	end
	local list = {}
	for _, p in ipairs(tpOrdered) do
		if tpChecked[p] and p.Parent then list[#list + 1] = p end
	end
	if #list == 0 then
		notify("Check objects in the list first")
		return
	end
	tourRunning = true
	tourButton.Text = "■  Stop tour"
	task.spawn(function()
		repeat
			for _, p in ipairs(list) do
				if not tourRunning then break end
				local t = teleportTo(p)
				if t then t.Completed:Wait() end
				local waited = 0
				while tourRunning and waited < S.tourDelay do
					task.wait(0.1)
					waited += 0.1
				end
			end
		until not tourRunning or not S.tourLoop
		stopTour()
	end)
end)

resetButton.MouseButton1Click:Connect(function()
	if rescanBusy then return end
	if not scriptOn then notify("Turn on Script power first") return end
	rescanBusy = true
	notify("Rescanning map…", true)

	for part in pairs(selected) do
		if not part:IsDescendantOf(Workspace) then onLost(part) end
	end

	local restored = 0
	if #pending > 0 then
		local all = Workspace:GetDescendants()
		for i, inst in ipairs(all) do
			if #pending == 0 then break end
			if inst:IsA("BasePart") and tryRefind(inst) then
				restored += 1
			end
			if i % 3000 == 0 then task.wait() end
		end
	end

	local hadLucky = luckySearched
	clearLucky()
	rescanBusy = false
	if hadLucky then
		runLucky()
	end
	scheduleRefresh()
	notify(("Map rescanned\n%d restored · %d waiting"):format(restored, #pending), true)
end)

	if refreshQueued then return end
	refreshQueued = true
	task.defer(function()
		refreshQueued = false
		refreshSelectedList()
		refreshTeleportList()
		updateBadges()
		for part in pairs(luckyMap) do syncLuckyVisibility(part) end
		refreshStatus()
	end)
end

UserInputService.InputBegan:Connect(function(input, gameProcessed)
	if isPress(input) then
		pressValid = not gameProcessed
		pressPos = input.Position
	elseif input.KeyCode == TOGGLE_KEY and not gameProcessed then
		setOpen(not panelOpen)
	end
end)

UserInputService.InputEnded:Connect(function(input)
	if not isPress(input) or not pressValid or not pressPos then
		return
	end
	pressValid = false
	if not scriptOn or mode ~= "Select" then
		return
	end
	if (input.Position - pressPos).Magnitude > CLICK_MOVE_THRESHOLD then
		return
	end
	local target = mouse.Target
	if not target or not target:IsA("BasePart") or target:IsA("Terrain") then
		return
	end
	local char = player.Character
	if char and target:IsDescendantOf(char) then
		return
	end
	if selected[target] then
		removeParts({ target })
	else
		if selectPart(target, pickColor) then
			pushHistory({ added = { { part = target, color = pickColor } }, removed = {} })
		end
	end
end)
